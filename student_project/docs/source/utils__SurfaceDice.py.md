 561 | def compute_surface_distances(mask_gt, mask_pred, spacing_mm):
 562 |     """Compute closest distances from all surface points to the other surface.
 563 | 
 564 |     Finds all surface elements "surfels" in the ground truth mask `mask_gt` and
 565 |     the predicted mask `mask_pred`, computes their area in mm^2 and the distance
 566 |     to the closest point on the other surface. It returns two sorted lists of
 567 |     distances together with the corresponding surfel areas. If one of the masks
 568 |     is empty, the corresponding lists are empty and all distances in the other
 569 |     list are `inf`
 570 | 
 571 |     Args:
 572 |       mask_gt: 3-dim Numpy array of type bool. The ground truth mask.
 573 |       mask_pred: 3-dim Numpy array of type bool. The predicted mask.
 574 |       spacing_mm: 3-element list-like structure. Voxel spacing in x0, x1 and x2
 575 |           direction
 576 | 
 577 |     Returns:
 578 |       A dict with
 579 |       "distances_gt_to_pred": 1-dim numpy array of type float. The distances in mm
 580 |           from all ground truth surface elements to the predicted surface,
 581 |           sorted from smallest to largest
 582 |       "distances_pred_to_gt": 1-dim numpy array of type float. The distances in mm
 583 |           from all predicted surface elements to the ground truth surface,
 584 |           sorted from smallest to largest
 585 |       "surfel_areas_gt": 1-dim numpy array of type float. The area in mm^2 of
 586 |           the ground truth surface elements in the same order as
 587 |           distances_gt_to_pred
 588 |       "surfel_areas_pred": 1-dim numpy array of type float. The area in mm^2 of
 589 |           the predicted surface elements in the same order as
 590 |           distances_pred_to_gt
 591 | 
 592 |     """
 593 | 
 594 |     # compute the area for all 256 possible surface elements
 595 |     # (given a 2x2x2 neighbourhood) according to the spacing_mm
 596 |     neighbour_code_to_surface_area = np.zeros([256])
 597 |     for code in range(256):
 598 |         normals = np.array(neighbour_code_to_normals[code])
 599 |         sum_area = 0
 600 |         for normal_idx in range(normals.shape[0]):
 601 |             # normal vector
 602 |             n = np.zeros([3])
 603 |             n[0] = normals[normal_idx, 0] * spacing_mm[1] * spacing_mm[2]
 604 |             n[1] = normals[normal_idx, 1] * spacing_mm[0] * spacing_mm[2]
 605 |             n[2] = normals[normal_idx, 2] * spacing_mm[0] * spacing_mm[1]
 606 |             area = np.linalg.norm(n)
 607 |             sum_area += area
 608 |         neighbour_code_to_surface_area[code] = sum_area
 609 | 
 610 |     # compute the bounding box of the masks to trim
 611 |     # the volume to the smallest possible processing subvolume
 612 |     mask_all = mask_gt | mask_pred
 613 |     bbox_min = np.zeros(3, np.int64)
 614 |     bbox_max = np.zeros(3, np.int64)
 615 | 
 616 |     # max projection to the x0-axis
 617 |     proj_0 = np.max(np.max(mask_all, axis=2), axis=1)
 618 |     idx_nonzero_0 = np.nonzero(proj_0)[0]
 619 |     if len(idx_nonzero_0) == 0:
 620 |         return {
 621 |             "distances_gt_to_pred": np.array([]),
 622 |             "distances_pred_to_gt": np.array([]),
 623 |             "surfel_areas_gt": np.array([]),
 624 |             "surfel_areas_pred": np.array([]),
 625 |         }
 626 | 
 627 |     bbox_min[0] = np.min(idx_nonzero_0)
 628 |     bbox_max[0] = np.max(idx_nonzero_0)
 629 | 
 630 |     # max projection to the x1-axis
 631 |     proj_1 = np.max(np.max(mask_all, axis=2), axis=0)
 632 |     idx_nonzero_1 = np.nonzero(proj_1)[0]
 633 |     bbox_min[1] = np.min(idx_nonzero_1)
 634 |     bbox_max[1] = np.max(idx_nonzero_1)
 635 | 
 636 |     # max projection to the x2-axis
 637 |     proj_2 = np.max(np.max(mask_all, axis=1), axis=0)
 638 |     idx_nonzero_2 = np.nonzero(proj_2)[0]
 639 |     bbox_min[2] = np.min(idx_nonzero_2)
 640 |     bbox_max[2] = np.max(idx_nonzero_2)
 641 | 
 642 |     #  print("bounding box min = {}".format(bbox_min))
 643 |     #  print("bounding box max = {}".format(bbox_max))
 644 | 
 645 |     # crop the processing subvolume.
 646 |     # we need to zeropad the cropped region with 1 voxel at the lower,
 647 |     # the right and the back side. This is required to obtain the "full"
 648 |     # convolution result with the 2x2x2 kernel
 649 |     cropmask_gt = np.zeros((bbox_max - bbox_min) + 2, np.uint8)
 650 |     cropmask_pred = np.zeros((bbox_max - bbox_min) + 2, np.uint8)
 651 | 
 652 |     cropmask_gt[0:-1, 0:-1, 0:-1] = mask_gt[
 653 |         bbox_min[0] : bbox_max[0] + 1,
 654 |         bbox_min[1] : bbox_max[1] + 1,
 655 |         bbox_min[2] : bbox_max[2] + 1,
 656 |     ]
 657 | 
 658 |     cropmask_pred[0:-1, 0:-1, 0:-1] = mask_pred[
 659 |         bbox_min[0] : bbox_max[0] + 1,
 660 |         bbox_min[1] : bbox_max[1] + 1,
 661 |         bbox_min[2] : bbox_max[2] + 1,
 662 |     ]
 663 | 
 664 |     # compute the neighbour code (local binary pattern) for each voxel
 665 |     # the resultsing arrays are spacially shifted by minus half a voxel in each axis.
 666 |     # i.e. the points are located at the corners of the original voxels
 667 |     kernel = np.array([[[128, 64], [32, 16]], [[8, 4], [2, 1]]])
 668 |     neighbour_code_map_gt = scipy.ndimage.filters.correlate(
 669 |         cropmask_gt.astype(np.uint8), kernel, mode="constant", cval=0
 670 |     )
 671 |     neighbour_code_map_pred = scipy.ndimage.filters.correlate(
 672 |         cropmask_pred.astype(np.uint8), kernel, mode="constant", cval=0
 673 |     )
 674 | 
 675 |     # create masks with the surface voxels
 676 |     borders_gt = (neighbour_code_map_gt != 0) & (neighbour_code_map_gt != 255)
 677 |     borders_pred = (neighbour_code_map_pred != 0) & (neighbour_code_map_pred != 255)
 678 | 
 679 |     # compute the distance transform (closest distance of each voxel to the surface voxels)
 680 |     if borders_gt.any():
 681 |         distmap_gt = scipy.ndimage.morphology.distance_transform_edt(
 682 |             ~borders_gt, sampling=spacing_mm
 683 |         )
 684 |     else:
 685 |         distmap_gt = np.Inf * np.ones(borders_gt.shape)
 686 | 
 687 |     if borders_pred.any():
 688 |         distmap_pred = scipy.ndimage.morphology.distance_transform_edt(
 689 |             ~borders_pred, sampling=spacing_mm
 690 |         )
 691 |     else:
 692 |         distmap_pred = np.Inf * np.ones(borders_pred.shape)
 693 | 
 694 |     # compute the area of each surface element
 695 |     surface_area_map_gt = neighbour_code_to_surface_area[neighbour_code_map_gt]
 696 |     surface_area_map_pred = neighbour_code_to_surface_area[neighbour_code_map_pred]
 697 | 
 698 |     # create a list of all surface elements with distance and area
 699 |     distances_gt_to_pred = distmap_pred[borders_gt]
 700 |     distances_pred_to_gt = distmap_gt[borders_pred]
 701 |     surfel_areas_gt = surface_area_map_gt[borders_gt]
 702 |     surfel_areas_pred = surface_area_map_pred[borders_pred]
 703 | 
 704 |     # sort them by distance
 705 |     if distances_gt_to_pred.shape != (0,):
 706 |         sorted_surfels_gt = np.array(sorted(zip(distances_gt_to_pred, surfel_areas_gt)))
 707 |         distances_gt_to_pred = sorted_surfels_gt[:, 0]
 708 |         surfel_areas_gt = sorted_surfels_gt[:, 1]
 709 | 
 710 |     if distances_pred_to_gt.shape != (0,):
 711 |         sorted_surfels_pred = np.array(
 712 |             sorted(zip(distances_pred_to_gt, surfel_areas_pred))
 713 |         )
 714 |         distances_pred_to_gt = sorted_surfels_pred[:, 0]
 715 |         surfel_areas_pred = sorted_surfels_pred[:, 1]
 716 | 
 717 |     return {
 718 |         "distances_gt_to_pred": distances_gt_to_pred,
